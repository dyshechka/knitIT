138void GraphicsViewZoom::gentleZoom(qreal factor)
139{
140 // A zero, negative or non-finite factor (a glitchy trackpad pinch event can produce one) would flip
141 // or collapse the whole view: the pattern turns upside down and the millimeter grid disappears.
142 if (not qIsFinite(factor) || factor <= 0)
143 {
144 return;
145 }
146
147 // If the view transform is already broken (flipped or degenerate), return it to a normal 100% scale.
148 const QTransform current = m_view->transform();
149 if (not qIsFinite(current.m11()) || current.m11() <= 0 || not qIsFinite(current.m22()) || current.m22() <= 0)
150 {
151 m_view->resetTransform();
152 }
153
154 // One event must not zoom too much at once, and the result must stay inside the allowed scale range.
155 const qreal m11 = m_view->transform().m11();
156 factor = qBound(0.5, factor, 2.0);
157 factor = qBound(VMainGraphicsView::MinScale() / m11, factor, VMainGraphicsView::MaxScale() / m11);
158
159 // See issue #532: Unexpected error occurs when zoom out image.
160 // factor > 1 for zoomIn and factor < 1 for zoomOut.
161 if (qAbs(factor - 1.0) > 1e-4)
162 {
163 m_view->scale(factor, factor);
164 if (factor < 1)
165 {
166 // Because QGraphicsView centers the picture when it's smaller than the view. And QGraphicsView's scrolls
167 // boundaries don't allow to put any picture point at any viewport position we will provide fictive scene
168 // size. Temporary and bigger than view, scene size will help position an image under cursor.
169 fictiveSceneRect(m_view->scene(), m_view);
170 }
171 m_view->centerOn(targetScenePos);
172 QPointF deltaViewPos = targetViewPos - QPointF(m_view->viewport()->width() / 2.0,
173 m_view->viewport()->height() / 2.0);
174 QPointF viewCenter = m_view->mapFromScene(targetScenePos) - deltaViewPos;
175 m_view->centerOn(m_view->mapToScene(viewCenter.toPoint()));
176 // In the end we just set correct scene size
177 VMainGraphicsScene *currentScene = qobject_cast<VMainGraphicsScene *>(m_view->scene());
178 SCASSERT(currentScene)
179 currentScene->setCurrentTransform(m_view->transform());
180 VMainGraphicsView::NewSceneRect(m_view->scene(), m_view);
181 emit zoomed();
182 }
183}

211/**
212 * @brief growSceneRect enlarges the scene rect around the visible area by half a view in every direction.
213 * The scene rect is only recalculated on zoom and on pattern changes, so when the user scrolled to its edge the
214 * scroll bars had no more range and the pattern "froze" (the trackpad did nothing) until the next zoom.
215 */
216void GraphicsViewZoom::growSceneRect()
217{
218 QGraphicsScene *sc = m_view->scene();
219 if (sc == nullptr)
220 {
221 return;
222 }
223
224 const QRectF viewRect = QRectF(m_view->mapToScene(0, 0),
225 m_view->mapToScene(m_view->viewport()->width(),
226 m_view->viewport()->height())).normalized();
227 QRectF grown = viewRect.adjusted(-viewRect.width() / 2., -viewRect.height() / 2.,
228 viewRect.width() / 2., viewRect.height() / 2.);
229
230 // Do not let the pattern get lost: the field may extend only a quarter of the view size beyond the pattern.
231 VMainGraphicsScene *currentScene = qobject_cast<VMainGraphicsScene *>(sc);
232 if (currentScene != nullptr)
233 {
234 const QRectF itemsRect = currentScene->visibleItemsBoundingRect();
235 if (not itemsRect.isNull())
236 {
237 const qreal marginX = viewRect.width() * 0.25;
238 const qreal marginY = viewRect.height() * 0.25;
239 const QRectF allowed = itemsRect.adjusted(-marginX, -marginY, marginX, marginY);
240 grown = grown.intersected(allowed);
241 }
242 }
243 const QRectF newRect = sc->sceneRect().united(grown);
244 if (newRect != sc->sceneRect())
245 {
246 sc->setSceneRect(newRect);
247 }
248}

251void GraphicsViewZoom::verticalScrollingTime(qreal x)
252{
253 Q_UNUSED(x)
254 // Try to adapt scrolling to speed of rotating mouse wheel and scale factor
255 // Value of _numScheduledScrollings is too short, so we scale the value
256
257 qreal scroll = static_cast<qreal>(qAbs(m_numScheduledVerticalScrollings))*(10. + 10./m_view->transform().m22())
258 /(static_cast<qreal>(m_duration)/static_cast<qreal>(m_updateInterval));
259
260 if (qAbs(scroll) < 1)
261 {
262 scroll = 1;
263 }
264
265 if (m_numScheduledVerticalScrollings > 0)
266 {
267 scroll = scroll * -1;
268 }
269 QScrollBar *bar = m_view->verticalScrollBar();
270 const int target = qRound(bar->value() + scroll);
271 if (target < bar->minimum() || target > bar->maximum())
272 {
273 growSceneRect();
274 }
275 bar->setValue(target);
276}

306//---------------------------------------------------------------------------------------------------------------------
307void GraphicsViewZoom::animFinished()
308{
309 m_numScheduledVerticalScrollings = 0;
310 verticalScrollAnim->stop();
311
312 /*
313 * In moust cases cursor position on view doesn't change, but for scene after scrolling position will be different.
314 * We are goint to check changes and save new value.
315 * If don't do that we will zoom using old value cursor position on scene. It is not what we expect.
316 * Almoust the same we do in method GraphicsViewZoom::eventFilter.
317 */
318 const QPoint pos = m_view->mapFromGlobal(QCursor::pos());
319 const QPointF delta = targetScenePos - m_view->mapToScene(pos);
320 if (qAbs(delta.x()) > 5 || qAbs(delta.y()) > 5)
321 {
322 targetViewPos = pos;
323 targetScenePos = m_view->mapToScene(pos);
324 }
325}

328bool GraphicsViewZoom::eventFilter(QObject *object, QEvent *event)
329{
330 m_modifiers = (qApp->Settings()->getZoomModKey()) ? Qt::ControlModifier : Qt::NoModifier;
331 const qreal m_zoomSpeedFactor = (qreal(qApp->Settings()->getZoomSpeedFactor())/10000) + 1.0004;
332
333 if (event->type() == QEvent::MouseMove)
334 {
335 /*
336 * Here we are saving cursor position on view and scene.
337 * This data need for gentleZoom().
338 * Almoust the same we do in method GraphicsViewZoom::animFinished.
339 */
340 QMouseEvent* mouse_event = static_cast<QMouseEvent*>(event);
341 QPointF delta = targetViewPos - mouse_event->pos();
342 if (qAbs(delta.x()) > 5 || qAbs(delta.y()) > 5)
343 {
344 targetViewPos = mouse_event->pos();
345 targetScenePos = m_view->mapToScene(mouse_event->pos());
346 }
347 return false;
348 }
349 else if (event->type() == QEvent::Wheel)
350 {
351 QWheelEvent* wheel_event = static_cast<QWheelEvent*>(event);
352 SCASSERT(wheel_event != nullptr)
353 if (QApplication::keyboardModifiers() == m_modifiers)
354 {
355 if (wheel_event->angleDelta().y() != 0)
356 {
357 const double angle = wheel_event->angleDelta().y();
358 const double factor = qPow(m_zoomSpeedFactor, angle);
359 qCDebug(vGraphicsViewZoom, "zoom Factor =%f \n", factor);
360 gentleZoom(factor);
361 return true;
362 }
363 }
364 else
365 {
366 // A real-world two-finger pinch rarely has zero drift -- the slight incidental
367 // motion often also arrives here as a wheel event alongside the pinch gesture
368 // itself (see pinchTriggered()/m_pinchInProgress). Starting a scroll animation
369 // from that drift would fight the pinch's own zoom (they'd both be adjusting the
370 // view at once) and can make the zoom look like it isn't working, so ignore wheel
371 // input entirely while a pinch is active.
372 if (m_pinchInProgress)
373 {
374 // A pinch that never reported its end must not block scrolling forever.
375 if (m_pinchLastEvent.isValid() && m_pinchLastEvent.elapsed() < 150)
376 {
377 return true;
378 }
379 m_pinchInProgress = false;
380 }
381
382 // A trackpad reports a genuine two-axis swipe (pixelDelta/angleDelta both
383 // populated) -- pick whichever axis carries the dominant motion so a left-right
384 // swipe scrolls horizontally on its own, without needing Shift. Shift still forces
385 // horizontal scrolling, for a plain mouse wheel that only ever reports a vertical
386 // delta.
387 const int absX = qMax(qAbs(wheel_event->pixelDelta().x()), qAbs(wheel_event->angleDelta().x()) / 8);
388 const int absY = qMax(qAbs(wheel_event->pixelDelta().y()), qAbs(wheel_event->angleDelta().y()) / 8);
389
390 if (QApplication::keyboardModifiers() == Qt::ShiftModifier || absX > absY)
391 {
392 return startHorizontalScrollings(wheel_event);
393 }
394 else
395 {
396 return startVerticalScrollings(wheel_event);
397 }
398 }
399 }
400 else if (event->type() == QEvent::Gesture)
401 {
402 return gestureEvent(static_cast<QGestureEvent*>(event));
403 }
404
405 return QObject::eventFilter(object, event);
406}

427void GraphicsViewZoom::panTriggered(QPanGesture *gesture)
428{
429#ifndef QT_NO_CURSOR
430 switch (gesture->state())
431 {
432 case Qt::GestureStarted:
433 case Qt::GestureUpdated:
434 m_view->viewport()->setCursor(Qt::SizeAllCursor);
435 break;
436 default:
437 m_view->viewport()->setCursor(Qt::ArrowCursor);
438 }
439#endif
440 QPointF delta = gesture->delta();
441 qCDebug(vGraphicsViewZoom) << "panTriggered():" << gesture;
442 horizontalOffset += delta.x();
443 verticalOffset += delta.y();
444 //m_view->scrollContentsBy(horizontalOffset, verticalOffset);
445}

447void GraphicsViewZoom::pinchTriggered(QPinchGesture *gesture)
448{
449 // See m_pinchInProgress: while true, the wheel-event handler in eventFilter() ignores
450 // incidental drift instead of scrolling, so it doesn't fight this gesture's own zoom.
451 m_pinchInProgress = (gesture->state() == Qt::GestureStarted || gesture->state() == Qt::GestureUpdated);
452 m_pinchLastEvent.restart();
453
454 if (gesture->state() == Qt::GestureStarted)
455 {
456 m_pinchPendingFactor = 1.0;
457 m_pinchTimer.invalidate();
458 }
459
460 QPinchGesture::ChangeFlags flags = gesture->changeFlags();
461 if (flags & QPinchGesture::ScaleFactorChanged)
462 {
463 // gesture->scaleFactor() is the incremental change since the *previous* gesture
464 // update -- exactly what gentleZoom() expects to multiply into the current view scale
465 // on every call. gesture->lastScaleFactor() is one step older (the increment that was
466 // current *before* this update), so using it here fed gentleZoom() stale data. See:
467 // https://doc.qt.io/qt-5/qpinchgesture.html#scaleFactor-prop
468 const qreal currentScaleFactor = gesture->scaleFactor();
469 if (qIsFinite(currentScaleFactor) && currentScaleFactor > 0)
470 {
471 m_pinchPendingFactor *= currentScaleFactor;
472 }
473 }
474
475 // A trackpad sends pinch updates much faster than the view can redraw (every step moves the
476 // scene rect and repaints the grid and every item), which makes zooming heavy. So merge updates
477 // and apply them at most every ~25 ms; the last bit is applied when the gesture ends.
478 const bool finished = (gesture->state() == Qt::GestureFinished || gesture->state() == Qt::GestureCanceled);
479 if (not qFuzzyCompare(m_pinchPendingFactor, 1.0)
480 && (finished || not m_pinchTimer.isValid() || m_pinchTimer.elapsed() >= 25))
481 {
482 const qreal total = m_pinchPendingFactor;
483 m_pinchPendingFactor = 1.0;
484 m_pinchTimer.restart();
485 gentleZoom(total);
486 }
487}

490void GraphicsViewZoom::fictiveSceneRect(QGraphicsScene *sc, QGraphicsView *view)
491{
492 SCASSERT(sc != nullptr)
493 SCASSERT(view != nullptr)
494
495 //Calculate view rect
496 //to receive the currently visible area, map the widgets bounds to the scene
497 const QPointF a = view->mapToScene(0, 0 );
498 const QPointF b = view->mapToScene(view->viewport()->width(), view->viewport()->height());
499 QRectF viewRect = QRectF( a, b );
500
501 //Scale view
502 QLineF topLeftRay(viewRect.center(), viewRect.topLeft());
503 topLeftRay.setLength(topLeftRay.length()*2);
504
505 QLineF bottomRightRay(viewRect.center(), viewRect.bottomRight());
506 bottomRightRay.setLength(bottomRightRay.length()*2);
507
508 viewRect = QRectF(topLeftRay.p2(), bottomRightRay.p2());
509
510 //Calculate scene rect
511 const QRectF sceneRect = sc->sceneRect();
512
513 //Unite two rects
514 const QRectF newRect = sceneRect.united(viewRect);
515
516 if (newRect != sceneRect)
517 {
518 sc->setSceneRect(newRect);
519 }
520}

523bool GraphicsViewZoom::startVerticalScrollings(QWheelEvent *wheel_event)
524{
525 SCASSERT(wheel_event != nullptr)
526
527 const QPoint numPixels = wheel_event->pixelDelta();
528 const QPoint numDegrees = wheel_event->angleDelta() / 8;
529 int numSteps;
530
531 if (not numPixels.isNull())
532 {
533 numSteps = numPixels.y();
534 }
535 else if (not numDegrees.isNull())
536 {
537 numSteps = numDegrees.y() / 15;
538 }
539 else
540 {
541 return true;//Just ignore
542 }
543
544 m_numScheduledVerticalScrollings += numSteps;
545 if (m_numScheduledVerticalScrollings * numSteps < 0)
546 { // if user moved the wheel in another direction, we reset previously scheduled scalings
547 m_numScheduledVerticalScrollings = numSteps;
548 }
549
550 m_numScheduledVerticalScrollings *= qint32(qApp->Settings()->getScrollSpeedFactor()/10.0);
551
552 if (verticalScrollAnim->state() != QTimeLine::Running)
553 {
554 verticalScrollAnim->start();
555 }
556 return true;
557}

559//---------------------------------------------------------------------------------------------------------------------
560bool GraphicsViewZoom::startHorizontalScrollings(QWheelEvent *wheel_event)
561{
562 SCASSERT(wheel_event != nullptr)
563
564 const QPoint numPixels = wheel_event->pixelDelta();
565 const QPoint numDegrees = wheel_event->angleDelta() / 8;
566 int numSteps;
567
568 if (not numPixels.isNull())
569 {
570 // Prefer the event's own horizontal component (a real trackpad left-right swipe);
571 // fall back to .y() for the Shift+mouse-wheel convention, where a wheel that only
572 // ever reports a vertical delta is being reinterpreted as horizontal because Shift
573 // is held.
574 numSteps = (numPixels.x() != 0) ? numPixels.x() : numPixels.y();
575 }
576 else if (not numDegrees.isNull())
577 {
578 numSteps = (numDegrees.x() != 0) ? numDegrees.x() / 15 : numDegrees.y() / 15;
579 }
580 else
581 {
582 return true;//Just ignore
583 }
584
585 m_numScheduledHorizontalScrollings += numSteps;
586 if (m_numScheduledHorizontalScrollings * numSteps < 0)
587 { // if user moved the wheel in another direction, we reset previously scheduled scalings
588 m_numScheduledHorizontalScrollings = numSteps;
589 }
590
591 m_numScheduledHorizontalScrollings *= qint32(qApp->Settings()->getScrollSpeedFactor()/10.0);
592
593 if (horizontalScrollAnim->state() != QTimeLine::Running)
594 {
595 horizontalScrollAnim->start();
596 }
597 return true;
598}

651namespace

653 struct MillimeterGridLevel
654 {
655 qreal stepMm;
656 int alpha; // 0-255 -- increases with coarseness so overlapping levels stay distinct.
657 };
658
659 // Seven fixed real-world steps (the 2 mm / 2 cm / 20 cm series were dropped to make drawing lighter), finest first -- Марта's chosen list, a fixed, predictable set
660 // rather than the original version's open-ended zoom-adaptive ladder. Each level is
661 // independently skipped once it's zoomed out past minGridSpacingPx below (an unreadable
662 // smudge) -- so however many of these ten currently clear that bar is exactly how many draw;
663 // there's no separate "pick 3" step. Going up to 1000 mm (1 m), not stopping at 1 cm or 5 cm,
664 // matters at the view's own zoomed-out floor (VMainGraphicsView::MinScale(), currently 21.5%):
665 // 1 cm was only barely above the smudge cutoff there (about 6.8px of the 6px minimum), so on
666 // real hardware it could disappear entirely; 1000 mm clears it with a wide margin (about
667 // 680px), so the grid never goes fully blank at maximum zoom-out. See DrawMillimeterGrid().
668 static const MillimeterGridLevel millimeterGridLevels[] =
669 {
670 {1.0, 25}, // 1 mm
671 {5.0, 55}, // 5 mm
672 {10.0, 70}, // 1 cm
673 {50.0, 110}, // 5 cm
674 {100.0, 130}, // 10 cm
675 {500.0, 170}, // 50 cm
676 {1000.0, 190}, // 1 m
677 };
678 static const int millimeterGridLevelCount =
679 static_cast<int>(sizeof(millimeterGridLevels) / sizeof(millimeterGridLevels[0]));
680
681 // Below this on-screen spacing (in device-independent pixels) a level is considered too
682 // dense to read -- drawing it anyway is exactly the "solid grey smudge" the spec warns about.
683 constexpr qreal minGridSpacingPx = 8.0;
684}

699//---------------------------------------------------------------------------------------------------------------------
700/**
701 * @brief DrawMillimeterGrid draws the "millimetrovka" background grid at its six fixed
702 * real-world steps (millimeterGridLevels above), in real-world millimeters regardless of the
703 * pattern's display unit (mm/cm/inch). Each level is skipped on its own once zoomed out far
704 * enough that it would draw closer than minGridSpacingPx apart (an unreadable smudge),
705 * independently of the other levels -- there is no separate zoom-tracking needed, since
706 * drawBackground() is already re-invoked on every repaint and repaints already follow
707 * signalZoomScaleChanged().
708 */
709void VMainGraphicsView::DrawMillimeterGrid(QPainter *painter, const QRectF &rect) const
710{
711 const qreal scaleFactor = transform().m11();
712 if (scaleFactor <= 0)
713 {
714 return;
715 }
716
717 // Scene coordinates are always expressed in the app's fixed 96 DPI reference unit
718 // (see ToPixel()/UnitConvertor() in def.cpp), independent of the pattern's display unit --
719 // so this is the one conversion needed to know how many scene units make up 1 real mm.
720 const qreal mmToScene = UnitConvertor(1.0, Unit::Mm, Unit::Px);
721 const qreal pixelsPerMm = scaleFactor * mmToScene;
722 if (pixelsPerMm <= 0)
723 {
724 return;
725 }
726
727 painter->save();
728
729 // Antialiasing ON for the grid, even though that softens each line slightly instead of a
730 // razor-crisp single device pixel. The real-world step sizes essentially never land on a
731 // whole number of device pixels at an arbitrary zoom percentage (e.g. 123%) -- with
732 // antialiasing off, each line's on/off pixel snapping rounds independently, so the visual
733 // gap between neighboring lines alternates between whatever pixel counts are nearby instead
734 // of staying constant, which is what actually made the grid look uneven across a wide zoom
735 // range (not just one particular percentage) even after fixing the double-painted-line issue
736 // below. With antialiasing on, a line at a fractional device position is drawn as smooth
737 // partial coverage instead of snapping unevenly, so spacing reads as uniform again.
738 painter->setRenderHint(QPainter::Antialiasing, true);
739
740 // Steel-blue, echoing the classic "миллиметровка" graph-paper look and the app's own blue
741 // accent color -- increasing opacity from the finest to the boldest level. Drawn coarsest
742 // first, finest last, and each level skips any line position that a coarser (already drawn
743 // this frame) level already covers -- e.g. the origin, or any other round-number coordinate,
744 // is a multiple of several step sizes at once, so without this a position like that gets
745 // painted several times over and the stacked semi-transparent ink makes that one line look
746 // much bolder than its neighbors (reported as an uneven-looking grid, e.g. at 123% zoom).
747 qreal drawnStepsScene[millimeterGridLevelCount];
748 int drawnStepsCount = 0;
749
750 for (int i = millimeterGridLevelCount - 1; i >= 0; --i)
751 {
752 const MillimeterGridLevel &level = millimeterGridLevels[i];
753 if (level.stepMm * pixelsPerMm >= minGridSpacingPx)
754 {
755 DrawMillimeterGridLevel(painter, rect, level.stepMm * mmToScene, drawnStepsScene,
756 drawnStepsCount, QColor(70, 130, 180, level.alpha));
757 drawnStepsScene[drawnStepsCount++] = level.stepMm * mmToScene;
758 }
759 }
760
761 painter->restore();
762}
763
764//---------------------------------------------------------------------------------------------------------------------
765/**
766 * @brief DrawMillimeterGridLevel draws one level of the grid: parallel lines `stepScene` scene
767 * units apart, clipped to the currently visible `rect`, in a single batched QPainter::drawLines()
768 * call for performance (this runs on every repaint, including continuous zoom/pan).
769 */
770void VMainGraphicsView::DrawMillimeterGridLevel(QPainter *painter, const QRectF &rect, qreal stepScene,
771 const qreal *excludeStepsScene, int excludeStepsCount,
772 const QColor &color) const
773{
774 if (stepScene <= 0 || not qIsFinite(stepScene)
775 || rect.width() / stepScene > 3000 || rect.height() / stepScene > 3000)
776 {
777 return;
778 }
779
780 // The grid is drawn in device pixels, without antialiasing, as 1-pixel wide filled rectangles.
781 // Antialiased cosmetic lines under the clip region QGraphicsView sets while repainting were the heaviest
782 // part of every repaint (about a third of the time on a 4K display), which made panning and zooming freeze.
783 const QTransform world = painter->worldTransform();
784 const qreal sx = world.m11();
785 const qreal sy = world.m22();
786 if (sx <= 0 || sy <= 0)
787 {
788 return;
789 }
790 const qreal dpr = (painter->device() != nullptr) ? painter->device()->devicePixelRatioF() : 1.0;
791 const qreal lineWidth = 1.0 / dpr;
792 const auto snap = [dpr](qreal v) -> qreal { return qRound(v * dpr) / dpr; };
793
794 const qreal topDev = world.map(QPointF(0, rect.top())).y();
795 const qreal bottomDev = world.map(QPointF(0, rect.bottom())).y();
796 const qreal leftDev = world.map(QPointF(rect.left(), 0)).x();
797 const qreal rightDev = world.map(QPointF(rect.right(), 0)).x();
798
799 const qreal left = qFloor(rect.left() / stepScene) * stepScene;
800 const qreal top = qFloor(rect.top() / stepScene) * stepScene;
801
802 // A position built as an exact multiple of stepScene (as every x/y below is, by
803 // construction) is also a multiple of a coarser, already-drawn step exactly when their
804 // ratio rounds to a whole number.
805 const auto alreadyDrawn = [excludeStepsScene, excludeStepsCount](qreal pos) -> bool
806 {
807 for (int i = 0; i < excludeStepsCount; ++i)
808 {
809 const qreal exclStep = excludeStepsScene[i];
810 if (exclStep <= 0)
811 {
812 continue;
813 }
814 const qreal ratio = pos / exclStep;
815 if (qAbs(ratio - qRound(ratio)) < 1e-6)
816 {
817 return true;
818 }
819 }
820 return false;
821 };
822
823 painter->save();
824 painter->setRenderHint(QPainter::Antialiasing, false);
825 painter->setWorldTransform(QTransform());
826
827 for (qreal x = left; x < rect.right(); x += stepScene)
828 {
829 if (!alreadyDrawn(x))
830 {
831 const qreal xDev = snap(world.map(QPointF(x, 0)).x());
832 painter->fillRect(QRectF(xDev, topDev, lineWidth, bottomDev - topDev), color);
833 }
834 }
835 for (qreal y = top; y < rect.bottom(); y += stepScene)
836 {
837 if (!alreadyDrawn(y))
838 {
839 const qreal yDev = snap(world.map(QPointF(0, y)).y());
840 painter->fillRect(QRectF(leftDev, yDev, rightDev - leftDev, lineWidth), color);
841 }
842 }
843
844 painter->restore();
845}
