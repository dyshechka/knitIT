86/**
87 * @brief eval calculate formula.
88 *
89 * First we try eval expression without adding variables. If it fail, we take tokens from expression and add variables
90 * to parser and try again.
91 *
92 * @param formula string of formula.
93 * @return value of formula.
94 */
95qreal Calculator::EvalFormula(const QHash<QString, QSharedPointer<VInternalVariable>> *vars, const QString &formula)
96{
97 // Parser doesn't know any variable on this stage. So, we just use variable factory that for each unknown variable
98 // set value to 0.
99 SetVarFactory(AddVariable, this);
100 SetSepForEval();//Reset separators options
101
102 SetExpr(formula);
103
104 qreal result = 0;
105 result = Eval();
106
107 QMap<int, QString> tokens = this->GetTokens();
108
109 // Remove "-" from tokens list if exist. If don't do that unary minus operation will broken.
110 RemoveAll(tokens, QStringLiteral("-"));
111
112 for (int i = 0; i < builInFunctions.size(); ++i)
113 {
114 if (tokens.isEmpty())
115 {
116 break;
117 }
118 RemoveAll(tokens, builInFunctions.at(i));
119 }
120
121 if (tokens.isEmpty())
122 {
123 return result; // We have found only numbers in expression.
124 }
125
126 // Add variables to parser because we have deal with expression with variables.
127 InitVariables(vars, tokens, formula);
128 return Eval();
129}

132/**
133 * @brief Calculator::GetUsedVariables list the variable/measurement names referenced by a
134 * formula. See the declaration in calculator.h for why this exists and what it reuses from
135 * EvalFormula() above.
136 */
137QStringList Calculator::GetUsedVariables(const QString &formula)
138{
139 // Parser doesn't know any variable at this stage either -- same var factory trick as
140 // EvalFormula() above, so an unknown name doesn't throw before we get a chance to list it.
141 SetVarFactory(AddVariable, this);
142 SetSepForEval();
143 SetExpr(formula);
144 Eval();
145
146 QMap<int, QString> tokens = this->GetTokens();
147
148 // Same cleanup as EvalFormula() above: strip the unary-minus token and built-in function
149 // names, leaving only genuine variable/measurement names.
150 RemoveAll(tokens, QStringLiteral("-"));
151 for (int i = 0; i < builInFunctions.size(); ++i)
152 {
153 if (tokens.isEmpty())
154 {
155 break;
156 }
157 RemoveAll(tokens, builInFunctions.at(i));
158 }
159
160 return tokens.values();
161}

164/**
165 * @brief Calculator::InitVariables add variables to parser.
166 *
167 * For optimization purpose we try don't add variables that we don't need.
168 *
169 * @param vars list of variables.
170 * @param tokens all tokens (measurements names, variables with lengths) that parser have found in expression.
171 * @param formula expression, need for throwing better error message.
172 */
173void Calculator::InitVariables(const QHash<QString, QSharedPointer<VInternalVariable> > *vars,
174 const QMap<int, QString> &tokens, const QString &formula)
175{
176 // Only used for the generic "Unexpected token" error this function used to throw below --
177 // no longer needed now that this has its own, more specific message. Left as a parameter
178 // (rather than changed to remove it) since it documents what the tokens/positions below are
179 // positions *into*, for anyone reading vs. calling this.
180 Q_UNUSED(formula);
181
182 QMap<int, QString>::const_iterator i = tokens.constBegin();
183 while (i != tokens.constEnd())
184 {
185 bool found = false;
186 if (vars->contains(i.value()))
187 {
188 DefineVar(i.value(), vars->value(i.value())->GetValue());
189 found = true;
190 }
191
192 if (found == false && builInFunctions.contains(i.value()))
193 {// We have found built-in function
194 found = true;
195 }
196
197 if (found == false)
198 {
199 // By this point the formula has already survived a full parse (EvalFormula()'s
200 // first Eval() pass, with a stub variable factory that accepts anything shaped
201 // like a name) -- so this token is never raw syntax garbage; garbage would already
202 // have thrown one of the parser's own "Unexpected ..." errors before this function
203 // ever ran. What lands here is always a token that parses fine AS A NAME but isn't
204 // one this formula actually has -- most often a mistyped, wrong-script (a Cyrillic
205 // letter standing in for its Latin look-alike, or the reverse), or no-longer-
206 // existing name. The generic parser wording ("Unexpected token") doesn't tell those
207 // two situations apart, which is exactly what made a mismatched name so hard to
208 // track down -- so this gets its own, more actionable message instead of reusing
209 // qmu::ecUNASSIGNABLE_TOKEN's generic text (see qmuparsererror.cpp).
210 throw qmu::QmuParserError(
211 QObject::tr("Variable \"$TOK$\" not found (position $POS$). Check that the name is "
212 "typed correctly (including letter case and keyboard layout) and that "
213 "it wasn't deleted or renamed."),
214 i.key(), i.value());
215 }
216 ++i;
217 }
218}
