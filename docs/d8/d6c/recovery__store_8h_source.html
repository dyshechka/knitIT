49/**
50 * @brief One editing window = one recovery session.
51 *
52 * Autosaved copies live in a single central folder (never next to the user's file), so a program crash,
53 * a read-only folder or a network drive cannot break them. The session keeps a lock file that the operating
54 * system releases automatically when the process dies; another start of the program sees "no owner" and
55 * knows the session was abandoned.
56 *
57 * Usage: start(); then for every autosave call beginVersion() -> write the document to the returned path ->
58 * commitVersion(). Call discard() after the document was saved normally or closed on purpose.
59 */
60class RecoverySession
61{
62public:
63 explicit RecoverySession(const QString &appName);
64 ~RecoverySession();
65
66 bool start();
67 bool isActive() const;
68
69 void setDocument(const QString &originalPath, const QString &title);
70
71 /// Path to write the next autosave to. The extension is always "autosave".
72 QString beginVersion();
73 /// Call after the file from beginVersion() has been written successfully; rotates versions.
74 bool commitVersion();
75 /// Removes every file of this session (normal exit or normal save).
76 void discard();
77
78 QString id() const;
79
80 static const int maxVersions = 3;
81
82 static QString recoveryDirPath();
83 static QList<AbandonedSession> findAbandoned(const QString &appName);
84 static void removeSession(const QString &id);
