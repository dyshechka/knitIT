46QString OldFormatPatternConverter::MeasurementsReference(const QString &patternPath)
47{
48 QFile file(patternPath);
49 if (!file.open(QIODevice::ReadOnly))
50 {
51 return QString();
52 }
53
54 QDomDocument doc;
55 const bool parsed = static_cast<bool>(doc.setContent(&file));
56 file.close();
57 if (!parsed)
58 {
59 return QString();
60 }
61
62 const QDomNodeList list = doc.documentElement().elementsByTagName(QStringLiteral("measurements"));
63 if (list.isEmpty())
64 {
65 return QString();
66 }
67 return list.at(0).toElement().text().trimmed();
68}

71OldFormatPatternConverter::Report OldFormatPatternConverter::Convert(const QString &sourcePath,
72 const QString &destPath,
73 const QHash<QString, QString> &renameMap,
74 const QString &newMeasurementsPath)
75{
76 Report report;
77
78 if (QFileInfo(sourcePath) == QFileInfo(destPath))
79 {
80 report.errorMessage = QStringLiteral("Файл результата должен отличаться от исходного файла.");
81 return report;
82 }
83
84 QFile inFile(sourcePath);
85 if (!inFile.open(QIODevice::ReadOnly))
86 {
87 report.errorMessage = QStringLiteral("Не удалось открыть исходную выкройку для чтения.");
88 return report;
89 }
90
91 VDomDocument doc;
92 QString parseError;
93 int errorLine = 0;
94 int errorColumn = 0;
95 const bool parsed = doc.setContent(&inFile, false, &parseError, &errorLine, &errorColumn);
96 inFile.close();
97 if (!parsed)
98 {
99 report.errorMessage = QStringLiteral("Файл не похож на выкройку (%1, строка %2).")
100 .arg(parseError).arg(errorLine);
101 return report;
102 }
103
104 QDomElement root = doc.documentElement();
105
106 // 1. Every attribute of every element: renamed measurement names + old rint() syntax.
107 // (QDomElement::elementsByTagName() has no wildcard, so walk the tree by hand.)
108 QList<QDomElement> pending;
109 pending.append(root);
110 while (!pending.isEmpty())
111 {
112 QDomElement element = pending.takeLast();
113 for (QDomNode child = element.firstChild(); !child.isNull(); child = child.nextSibling())
114 {
115 if (child.isElement())
116 {
117 pending.append(child.toElement());
118 }
119 }
120
121 const QDomNamedNodeMap attributes = element.attributes();
122
123 // Collect first: setAttribute() on the element being iterated would invalidate the map.
124 QList<QPair<QString, QString> > updates;
125 for (int a = 0; a < attributes.size(); ++a)
126 {
127 const QDomAttr attr = attributes.item(a).toAttr();
128 const QString oldValue = attr.value();
129 if (oldValue.isEmpty())
130 {
131 continue;
132 }
133
134 const QString renamed = OldFormatMeasurementsConverter::ApplyRenames(oldValue, renameMap);
135 QString finalValue = renamed;
136 if (renamed.contains(QStringLiteral("rint(")))
137 {
138 finalValue = OldFormatMeasurementsConverter::ConvertRintSyntax(renamed);
139 }
140
141 if (renamed != oldValue)
142 {
143 ++report.renamedAttributes;
144 }
145 if (finalValue != renamed)
146 {
147 ++report.rintConvertedCount;
148 }
149 if (finalValue != oldValue)
150 {
151 updates.append(qMakePair(attr.name(), finalValue));
152 }
153 }
154
155 for (const QPair<QString, QString> &update : qAsConst(updates))
156 {
157 element.setAttribute(update.first, update.second);
158 }
159 }
160
161 // 2. The reference to the measurements file.
162 if (!newMeasurementsPath.isEmpty())
163 {
164 const QDomNodeList refs = root.elementsByTagName(QStringLiteral("measurements"));
165 if (!refs.isEmpty())
166 {
167 QDomElement ref = refs.at(0).toElement();
168 while (ref.hasChildNodes())
169 {
170 ref.removeChild(ref.firstChild());
171 }
172 const QString relative = QDir(QFileInfo(destPath).absolutePath()).relativeFilePath(newMeasurementsPath);
173 ref.appendChild(doc.createTextNode(relative));
174 }
175 }
176
177 // 3. A note that the file went through the converter (an XML comment: the pattern schema does not allow
178 // unknown elements, comments are invisible to it).
179 const QString conversionComment = QStringLiteral(" Converted from the old format (\"@\" + old rint() syntax) "
180 "by Seamly2D v%1 on %2. ")
181 .arg(APP_VERSION_STR, QDate::currentDate().toString(Qt::ISODate));
182 const QDomComment conversionInfo = doc.createComment(conversionComment);
183 const QDomElement versionEl = root.firstChildElement(VDomDocument::TagVersion);
184 if (!versionEl.isNull())
185 {
186 root.insertAfter(conversionInfo, versionEl);
187 }
188 else
189 {
190 root.insertBefore(conversionInfo, root.firstChild());
191 }
192
193 QString saveError;
194 if (!doc.SaveDocument(destPath, saveError))
195 {
196 report.errorMessage = saveError.isEmpty() ? QStringLiteral("Не удалось сохранить итоговую выкройку.")
197 : saveError;
198 }
199 return report;
200}
