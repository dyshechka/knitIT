89//---------------------------------------------------------------------------------------------------------------------
90bool RecoverySession::start()
91{
92 if (isActive())
93 {
94 return true;
95 }
96
97 if (!QDir().mkpath(recoveryDirPath()))
98 {
99 return false;
100 }
101
102 m_lock.reset(new QLockFile(filePath(lockSuffix)));
103 // 0 = the lock is stale only when its owner process is dead, never because of its age
104 m_lock->setStaleLockTime(0);
105 if (!m_lock->tryLock(0))
106 {
107 m_lock.reset();
108 return false;
109 }
110 return true;
111}

136//---------------------------------------------------------------------------------------------------------------------
137bool RecoverySession::commitVersion()
138{
139 const QString fresh = filePath(QStringLiteral("new.") + versionSuffix);
140 if (!QFileInfo::exists(fresh))
141 {
142 return false;
143 }
144
145 const QString dir = recoveryDirPath();
146 // shift: v(n-1) -> v(n), ..., v1 -> v2
147 QFile::remove(dir + QLatin1Char('/') + versionName(m_id, maxVersions));
148 for (int i = maxVersions - 1; i >= 1; --i)
149 {
150 const QString from = dir + QLatin1Char('/') + versionName(m_id, i);
151 if (QFileInfo::exists(from))
152 {
153 QFile::rename(from, dir + QLatin1Char('/') + versionName(m_id, i + 1));
154 }
155 }
156
157 if (!QFile::rename(fresh, dir + QLatin1Char('/') + versionName(m_id, 1)))
158 {
159 QFile::remove(fresh);
160 return false;
161 }
162 return writeMeta();
163}

165//---------------------------------------------------------------------------------------------------------------------
166bool RecoverySession::writeMeta() const
167{
168 QJsonObject obj;
169 obj.insert(QStringLiteral("app"), m_app);
170 obj.insert(QStringLiteral("original"), m_original);
171 obj.insert(QStringLiteral("title"), m_title);
172 obj.insert(QStringLiteral("time"), QDateTime::currentDateTime().toString(Qt::ISODate));
173
174 QSaveFile file(filePath(metaSuffix));
175 if (!file.open(QIODevice::WriteOnly))
176 {
177 return false;
178 }
179 file.write(QJsonDocument(obj).toJson(QJsonDocument::Compact));
180 return file.commit();
181}

206void RecoverySession::removeSession(const QString &id)
207{
208 const QString dir = recoveryDirPath();
209 for (int i = 1; i <= maxVersions; ++i)
210 {
211 QFile::remove(dir + QLatin1Char('/') + versionName(id, i));
212 }
213 QFile::remove(dir + QLatin1Char('/') + id + QStringLiteral(".new.") + versionSuffix);
214 QFile::remove(dir + QLatin1Char('/') + id + QLatin1Char('.') + metaSuffix);
215 QFile::remove(dir + QLatin1Char('/') + id + QLatin1Char('.') + lockSuffix);

218//---------------------------------------------------------------------------------------------------------------------
219QList<AbandonedSession> RecoverySession::findAbandoned(const QString &appName)
220{
221 QList<AbandonedSession> result;
222
223 const QDir dir(recoveryDirPath());
224 const QStringList metas = dir.entryList(QStringList() << QStringLiteral("*.") + metaSuffix, QDir::Files);
225 for (const QString &metaName : metas)
226 {
227 const QString id = QFileInfo(metaName).completeBaseName();
228
229 // Is the owner still alive? tryLock succeeds when the lock is free or its owner process is dead.
230 QLockFile lock(dir.filePath(id + QLatin1Char('.') + lockSuffix));
231 lock.setStaleLockTime(0);
232 if (!lock.tryLock(0))
233 {
234 continue; // another window of the program is working on it right now
235 }
236
237 QFile metaFile(dir.filePath(metaName));
238 if (!metaFile.open(QIODevice::ReadOnly))
239 {
240 continue;
241 }
242 const QJsonObject obj = QJsonDocument::fromJson(metaFile.readAll()).object();
243 metaFile.close();
244 if (obj.value(QStringLiteral("app")).toString() != appName)
245 {
246 continue;
247 }
248
249 AbandonedSession session;
250 session.id = id;
251 session.app = appName;
252 session.originalPath = obj.value(QStringLiteral("original")).toString();
253 session.title = obj.value(QStringLiteral("title")).toString();
254 session.time = QDateTime::fromString(obj.value(QStringLiteral("time")).toString(), Qt::ISODate);
255 for (int i = 1; i <= maxVersions; ++i)
256 {
257 const QString path = dir.filePath(versionName(id, i));
258 if (QFileInfo::exists(path))
259 {
260 session.versions.append(path);
261 }
262 }
263 if (session.versions.isEmpty())
264 {
265 // nothing to restore: just clean up the leftovers silently
266 lock.unlock();
267 removeSession(id);
268 continue;
269 }
270 result.append(session);
271 }
272 return result;
273}
