47 // Any character the qmuparser formula grammar treats as part of an identifier (see
48 // QmuFormulaBase::InitCharSets() in qmuformulabase.cpp): a Unicode letter or digit, "_",
49 // "@", "#" or "'". Used to make sure a rename only ever matches a *whole* identifier token,
50 // never a substring of a longer one (so renaming "Р" can't accidentally touch "Рост").

56{
57 return c.isLetterOrNumber() || c == QLatin1Char('_') || c == QLatin1Char('@') ||
58 c == QLatin1Char('#') || c == QLatin1Char('\'');

62int OldFormatMeasurementsConverter::FindMatchingParen(const QString &s, int openIdx)
63{
64 int depth = 0;
65 for (int i = openIdx; i < s.size(); ++i)
66 {
67 if (s.at(i) == QLatin1Char('('))
68 {
69 ++depth;
70 }
71 else if (s.at(i) == QLatin1Char(')'))
72 {
73 --depth;
74 if (depth == 0)
75 {
76 return i;
77 }
78 }
79 }
80 return -1;
81}

83//---------------------------------------------------------------------------------------------------------------------
84QString OldFormatMeasurementsConverter::StripRedundantParens(const QString &expr)
85{
86 QString result = expr.trimmed();
87 while (result.size() >= 2 && result.at(0) == QLatin1Char('('))
88 {
89 const int close = FindMatchingParen(result, 0);
90 if (close == result.size() - 1)
91 {
92 result = result.mid(1, result.size() - 2).trimmed();
93 }
94 else
95 {
96 break;
97 }
98 }
99 return result;
100}

103QString OldFormatMeasurementsConverter::CollapseAtSign(const QString &name)
104{
105 if (name.startsWith(QLatin1String("@@")))
106 {
107 return name.mid(1); // "@@k" -> "@k": keep exactly one "@"
108 }
109 if (name.startsWith(QLatin1Char('@')))
110 {
111 return name.mid(1); // "@k" -> "k": remove entirely
112 }
113 return name;
114}

117/// @brief Converts every old-style rint((<expr>) * 10^n) / 10^n call in formula to the new
118/// rint(<expr>; n) syntax, bottom-up (a nested old-style call inside another one's argument is
119/// converted first). A rint(...) call that isn't immediately followed by a divide-by-the-same-
120/// value is left exactly as it was.
121QString OldFormatMeasurementsConverter::ConvertRintSyntax(const QString &formula)
122{
123 static const QRegularExpression multRe(QStringLiteral("^(.*)\\*\\s*([0-9]+)\\s*$"));
124 static const QRegularExpression powerOfTenRe(QStringLiteral("^10*$"));
125
126 QString out;
127 out.reserve(formula.size());
128 int i = 0;
129 const int n = formula.size();
130 while (i < n)
131 {
132 const bool looksLikeRintCall = (i + 5 <= n) && formula.at(i) == QLatin1Char('r') &&
133 formula.at(i + 1) == QLatin1Char('i') && formula.at(i + 2) == QLatin1Char('n') &&
134 formula.at(i + 3) == QLatin1Char('t') && formula.at(i + 4) == QLatin1Char('(');
135 const bool onBoundary = (i == 0) || !IsIdentChar(formula.at(i - 1));
136
137 if (looksLikeRintCall && onBoundary)
138 {
139 const int openIdx = i + 4; // index of '('
140 const int closeIdx = FindMatchingParen(formula, openIdx);
141 if (closeIdx == -1)
142 {
143 out += formula.at(i);
144 ++i;
145 continue;
146 }
147
148 const QString argsRaw = formula.mid(openIdx + 1, closeIdx - openIdx - 1);
149 const QString argsConverted = ConvertRintSyntax(argsRaw); // nested rint() first
150
151 bool converted = false;
152 const QRegularExpressionMatch mm = multRe.match(argsConverted);
153 if (mm.hasMatch())
154 {
155 QString innerExpr = mm.captured(1).trimmed();
156 const QString multStr = mm.captured(2);
157 if (powerOfTenRe.match(multStr).hasMatch()) // a power of ten: 1, 10, 100, 1000...
158 {
159 const QString after = formula.mid(closeIdx + 1);
160 const QRegularExpression afterRe(QStringLiteral("^\\s*/\\s*") + QRegularExpression::escape(multStr) +
161 QStringLiteral("(?![0-9])"));
162 const QRegularExpressionMatch am = afterRe.match(after);
163 if (am.hasMatch())
164 {
165 const int precision = multStr.size() - 1;
166 innerExpr = StripRedundantParens(innerExpr);
167 out += QStringLiteral("rint(%1; %2)").arg(innerExpr).arg(precision);
168 i = closeIdx + 1 + am.capturedLength(0);
169 converted = true;
170 }
171 }
172 }
173
174 if (!converted)
175 {
176 out += QLatin1String("rint(") + argsConverted + QLatin1Char(')');
177 i = closeIdx + 1;
178 }
179 continue;
180 }
181
182 out += formula.at(i);
183 ++i;
184 }
185 return out;
186}

189/// @brief Rewrites every whole-identifier-token occurrence of a renamed measurement name inside
190/// value (a formula) to its new name, in a single pass so a freshly-inserted replacement can
191/// never be re-matched by a later rename in the same formula.
192QString OldFormatMeasurementsConverter::ApplyRenames(const QString &value, const QHash<QString, QString> &renameMap)
193{
194 if (renameMap.isEmpty())
195 {
196 return value;
197 }
198
199 QStringList names = renameMap.keys();
200 std::sort(names.begin(), names.end(), [](const QString &a, const QString &b)
201 {
202 return a.size() > b.size();
203 });
204
205 QStringList escaped;
206 escaped.reserve(names.size());
207 for (const QString &nm : qAsConst(names))
208 {
209 escaped.append(QRegularExpression::escape(nm));
210 }
211
212 const QString pattern = QStringLiteral("(?<!%1)(?:%2)(?!%1)").arg(IdentBoundaryExclude, escaped.join(QLatin1Char('|')));
213 const QRegularExpression re(pattern, QRegularExpression::UseUnicodePropertiesOption);
214
215 QString result;
216 result.reserve(value.size());
217 int lastEnd = 0;
218 QRegularExpressionMatchIterator it = re.globalMatch(value);
219 while (it.hasNext())
220 {
221 const QRegularExpressionMatch m = it.next();
222 result += value.mid(lastEnd, m.capturedStart() - lastEnd);
223 result += renameMap.value(m.captured(0));
224 lastEnd = m.capturedEnd();
225 }
226 result += value.mid(lastEnd);
227 return result;
228}

231/// @brief Decides the final name for every measurement: "@"/"@@" collapsed per CollapseAtSign(),
232/// except that any measurement whose collapsed name would collide with another measurement's
233/// (collapsed or already-plain) name keeps its original name untouched, with a warning. Runs to
234/// a fixed point, since reverting one collision can in rare cases create another.
235void OldFormatMeasurementsConverter::ComputeRenameMap(const QStringList &names, QStringList &finalNames,
236 QStringList &warnings)
237{
238 const int n = names.size();
239 QVector<bool> reverted(n, false);
240
241 bool changed = true;
242 while (changed)
243 {
244 changed = false;
245
246 QStringList candidate;
247 candidate.reserve(n);
248 for (int i = 0; i < n; ++i)
249 {
250 candidate.append(reverted.at(i) ? names.at(i) : CollapseAtSign(names.at(i)));
251 }
252
253 QHash<QString, int> counts;
254 for (const QString &c : qAsConst(candidate))
255 {
256 ++counts[c];
257 }
258
259 for (int i = 0; i < n; ++i)
260 {
261 if (!reverted.at(i) && counts.value(candidate.at(i)) > 1)
262 {
263 reverted[i] = true;
264 changed = true;
265 }
266 }
267 }
268
269 finalNames.clear();
270 finalNames.reserve(n);
271 warnings.clear();
272 for (int i = 0; i < n; ++i)
273 {
274 const QString collapsed = CollapseAtSign(names.at(i));
275 if (reverted.at(i))
276 {
277 finalNames.append(names.at(i));
278 if (names.at(i) != collapsed)
279 {
280 warnings.append(QCoreApplication::translate("OldFormatMeasurementsConverter",
281 "«%1»: после удаления «@» это имя совпало бы с другой меркой в этом же файле "
282 "-- оставлено без изменений.").arg(names.at(i)));
283 }
284 }
285 else
286 {
287 finalNames.append(collapsed);
288 }
289 }
290}

293OldFormatMeasurementsConverter::Report OldFormatMeasurementsConverter::Convert(const QString &sourcePath,
294 const QString &destPath)
295{
296 Report report;
297
298 if (sourcePath == destPath)
299 {
300 report.errorMessage = QCoreApplication::translate("OldFormatMeasurementsConverter",
301 "Файл результата должен отличаться от исходного файла.");
302 return report;
303 }
304
305 QFile inFile(sourcePath);
306 if (!inFile.open(QIODevice::ReadOnly))
307 {
308 report.errorMessage = QCoreApplication::translate("OldFormatMeasurementsConverter",
309 "Не удалось открыть исходный файл для чтения.");
310 return report;
311 }
312
313 VDomDocument doc;
314 QString parseError;
315 int errorLine = 0;
316 int errorColumn = 0;
317 const bool parsed = doc.setContent(&inFile, false, &parseError, &errorLine, &errorColumn);
318 inFile.close();
319 if (!parsed)
320 {
321 report.errorMessage = QCoreApplication::translate("OldFormatMeasurementsConverter",
322 "Файл не похож на файл мерок (%1, строка %2).").arg(parseError).arg(errorLine);
323 return report;
324 }
325
326 QDomElement root = doc.documentElement();
327 const QDomNodeList mList = root.elementsByTagName(QStringLiteral("m"));
328 const int n = mList.size();
329
330 QStringList names;
331 names.reserve(n);
332 for (int i = 0; i < n; ++i)
333 {
334 names.append(mList.at(i).toElement().attribute(QStringLiteral("name")));
335 }
336
337 QStringList finalNames;
338 ComputeRenameMap(names, finalNames, report.collisionWarnings);
339
340 QHash<QString, QString> renameMap;
341 for (int i = 0; i < n; ++i)
342 {
343 if (names.at(i) != finalNames.at(i))
344 {
345 renameMap.insert(names.at(i), finalNames.at(i));
346 }
347 }
348 report.renamedCount = renameMap.size();
349 report.renameMap = renameMap;
350
351 for (int i = 0; i < n; ++i)
352 {
353 QDomElement m = mList.at(i).toElement();
354 m.setAttribute(QStringLiteral("name"), finalNames.at(i));
355
356 const QString oldValue = m.attribute(QStringLiteral("value"));
357 if (oldValue.isEmpty())
358 {
359 continue;
360 }
361
362 const QString renamedValue = ApplyRenames(oldValue, renameMap);
363 const QString finalValue = ConvertRintSyntax(renamedValue);
364
365 if (finalValue != renamedValue)
366 {
367 ++report.rintConvertedCount;
368 }
369 if (finalValue != oldValue)
370 {
371 m.setAttribute(QStringLiteral("value"), finalValue);
372 }
373 }
374
375 // Record that this file went through the converter, for future comparison. This has to be an
376 // XML *comment*, not an element: opening a .smis file runs it through IndividualSizeConverter,
377 // which validates it against a strict XSD (no wildcard/"any" content) via Xerces -- any element
378 // the schema doesn't declare (we tried a <conversion-info> element first) makes the file fail to
379 // open at all, with "no declaration found for element ...". A comment is invisible to that
380 // validation, same as the pre-existing "Measurements created with SeamlyMe v..." comment already
381 // at the top of the file. We don't touch that first comment or <version> itself -- <version>
382 // belongs to IndividualSizeConverter's own, unrelated format-migration logic, and the first
383 // comment is what MeasurementDoc::SaveDocument() rewrites on every save from inside the app.
384 const QString conversionComment = QStringLiteral(" Converted from the old format (\"@\" + old rint() syntax) "
385 "by SeamlyMe v%1 on %2. ")
386 .arg(APP_VERSION_STR, QDate::currentDate().toString(Qt::ISODate));
387 const QDomComment conversionInfo = doc.createComment(conversionComment);
388
389 const QDomElement versionEl = root.firstChildElement(VDomDocument::TagVersion);
390 if (!versionEl.isNull())
391 {
392 root.insertAfter(conversionInfo, versionEl);
393 }
394 else
395 {
396 root.insertBefore(conversionInfo, root.firstChild());
397 }
398
399 QString saveError;
400 if (!doc.SaveDocument(destPath, saveError))
401 {
402 report.errorMessage = saveError.isEmpty()
403 ? QCoreApplication::translate("OldFormatMeasurementsConverter", "Не удалось сохранить итоговый файл.")
404 : saveError;
405 return report;
406 }
407
408 return report;
409}
