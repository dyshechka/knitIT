84/// @brief setLabel set the text and the position of the label.
85/// @param text text, lines are separated by a new line character.
86/// @param center middle of the line in the coordinates of the parent item.
87/// @param offset how far the label is moved from the middle of the line.
88void VLineLengthLabel::setLabel(const QString &text, const QPointF &center, const QPointF &offset)
89{
90 prepareGeometryChange();
91 m_text = text;
92
93 // Same font and colors as the point names use. Read here, not at every repaint: settings are slow.
94 m_font = qApp->Settings()->getPointNameFont();
95 m_font.setPointSize(qApp->Settings()->getPointNameSize());
96 m_textColor = QColor(qApp->Settings()->getPointNameColor());
97 m_textColor.setAlpha(224);
98 m_leaderColor = QColor(qApp->Settings()->getPointNameColor());
99 m_leaderColor.setAlpha(128);
100
101 const QFontMetricsF metrics(m_font);
102 const QRectF textRect = metrics.boundingRect(QRectF(0, 0, 10000, 10000), Qt::AlignLeft, m_text);
103 m_rect = QRectF(-textRect.width() / 2.0 - labelPadding, -textRect.height() / 2.0 - labelPadding,
104 textRect.width() + 2 * labelPadding, textRect.height() + 2 * labelPadding);
105
106 m_shape = QPainterPath();
107 m_shape.addRect(m_rect);
108
109 m_center = center;
110 m_offset = offset;
111 setPos(m_center + m_offset);
112 refreshLeader();
113 update();
114}

136/// @brief refreshLeader work out the leader line from the label to the middle of its line (in the coordinates
137/// of the label). There is none if the label is close to the middle of the line.
138/// Must be called every time the position, the scale or the size of the label changes.
139void VLineLengthLabel::refreshLeader()
140{
141 m_hasLeader = false;
142 if (parentItem() == nullptr)
143 {
144 return;
145 }
146
147 const QPointF target = mapFromParent(m_center);
148 if (m_rect.contains(target))
149 {
150 return;
151 }
152
153 // The point where the ray from the label center to the target leaves the label.
154 const QLineF ray(m_rect.center(), target);
155 const QLineF edges[] = {QLineF(m_rect.topLeft(), m_rect.topRight()), QLineF(m_rect.topRight(), m_rect.bottomRight()),
156 QLineF(m_rect.bottomRight(), m_rect.bottomLeft()),
157 QLineF(m_rect.bottomLeft(), m_rect.topLeft())};
158 QPointF start = m_rect.center();
159 for (const QLineF &edge : edges)
160 {
161 QPointF crossing;
162 if (ray.intersects(edge, &crossing) == QLineF::BoundedIntersection)
163 {
164 start = crossing;
165 break;
166 }
167 }
168
169 m_leader = QLineF(start, target);
170 m_hasLeader = m_leader.length() > minLeaderLength;
171}

174void VLineLengthLabel::paint(QPainter *painter, const QStyleOptionGraphicsItem *option, QWidget *widget)
175{
176 Q_UNUSED(widget)
177
178 // Zoomed far out the text is smaller than a pixel: same shortcut as the point names have.
179 if (option->levelOfDetailFromTransform(painter->worldTransform()) < 0.1)
180 {
181 return;
182 }
183
184 // The same behaviour as the point names have: keep the size when the pattern is zoomed in,
185 // and shrink together with the pattern when it is zoomed out.
186 const qreal viewScale = sceneScale(scene());
187 const qreal wanted = viewScale > 1.0 ? 1.0 / viewScale : 1.0;
188 if (not qFuzzyCompare(scale(), wanted))
189 {
190 setScale(wanted);
191 refreshLeader();
192 }
193
194 painter->save();
195 painter->setRenderHint(QPainter::TextAntialiasing, true);
196
197 if (m_hasLeader)
198 {
199 QPen leaderPen(m_leaderColor, 1);
200 leaderPen.setCosmetic(true);
201 painter->setPen(leaderPen);
202 painter->drawLine(m_leader);
203 }
204
205 painter->setFont(m_font);
206 painter->setPen(m_textColor);
207 painter->drawText(m_rect, Qt::AlignCenter, m_text); // every line is centered, lines are split by '\n'
208 painter->restore();
209}

224//---------------------------------------------------------------------------------------------------------------------
225void VLineLengthLabel::mouseMoveEvent(QGraphicsSceneMouseEvent *event)
226{
227 if (not m_movable)
228 {
229 return;
230 }
231 prepareGeometryChange(); // the leader line follows the label
232
233 QPointF delta = event->scenePos() - m_pressScenePos;
234 if (parentItem() != nullptr)
235 {
236 delta = parentItem()->mapFromScene(event->scenePos()) - parentItem()->mapFromScene(m_pressScenePos);
237 }
238 setPos(m_pressPos + delta);
239 refreshLeader();
240 update(); // the leader line has just changed: draw the new one, wipe the old one
241}

253 m_moved(m_offset);
254 }
